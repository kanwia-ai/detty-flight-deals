  generate_digest(db_client, subscriber: dict) -> dict | None:
    """Generate a personalized digest for a single subscriber.

    Returns dict with {subject, html_body, plain_body, deal_count, teaser_count}
    or None if no content for this subscriber.
    """

    - If no pending deals, return None

    - Separate deals by type:
      - great_deals: [d for d in deals if d["tier"] in ("great", "good") and d.get("expired", 0) == 0]
      - teaser_deals: [d for d in deals if d["tier"] in ("wow", "mistake") or d.get("expired", 0) == 1]

    - Filter great_deals by subscriber metro:
      Import airport_matches_subscriber from subscriber.metro_groups
      filtered_great = [d for d in great_deals if airport_matches_subscriber(d["origin"], subscriber)]

    - Cap filtered_great at MAX_DIGEST_DEALS
    - Select up to MAX_FOMO_TEASERS from teaser_deals at random (import random)
      Use random.sample(teaser_deals, min(MAX_FOMO_TEASERS, len(teaser_deals)))

    - If no filtered_great and no selected teasers, return None (nothing for this subscriber)

    - Get subscriber info:
      name = subscriber.get("name", "")
      metro = subscriber.get("metro_group", "your area")

    - Build email content:
      Import build_weekly_digest_html, build_weekly_digest_plain, build_weekly_digest_subject from alert.templates

      subject = build_weekly_digest_subject(
        deal_count=len(filtered_great),
        best_dest=filtered_great[0].get("dest_name") if filtered_great else None,
        best_price=filtered_great[0].get("price_cents", 0) // 100 if filtered_great else None,
      )
      html_body = build_weekly_digest_html(name, filtered_great, selected_teasers, metro)
      plain_body = build_weekly_digest_plain(name, filtered_great, selected_teasers, metro)

    - Return {"subject": subject, "html_body": html_body, "plain_body": plain_body,
              "deal_count": len(filtered_great), "teaser_count": len(selected_teasers)}
