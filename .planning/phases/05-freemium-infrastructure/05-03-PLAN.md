<objective>
Build the alert routing system that dispatches deals to the correct subscribers based on tier and metro preferences, then wire it into deal_finder.py to replace the current "send everything to everyone" approach.

Purpose: This is the core freemium logic -- the router decides who gets what. Free users get Great deals queued for weekly digest. Premium/trial users get everything instantly. Mistake fares trigger SMS. This plan replaces the monolithic send_email() with tier-aware routing.

Output: subscriber/router.py (AlertRouter), subscriber/sms.py (Twilio SMS), updated deal_finder.py with tier-based routing.
</objective>

  route_deal(self, deal: dict) -> dict:
    """Route a single deal to the correct subscribers and delivery channels.

    Returns a dict with routing summary:
      {"instant_emails": N, "sms_sent": N, "digest_queued": bool, "skipped_limit": bool}
    """
    subscribers = self._load_subscribers()
    origin = deal.get("origin", "")
    tier = deal.get("tier", "").lower()
    is_mistake = deal.get("is_mistake_fare", False)
    result = {"instant_emails": 0, "sms_sent": 0, "digest_queued": False, "skipped_limit": False}

    # Determine deal type for routing
    is_premium_content = tier in ("wow",) or is_mistake
    is_free_content = tier in ("good", "great")

    # --- Premium/Trial instant delivery ---
    if is_premium_content or is_free_content:
      premium_subs = [s for s in subscribers
                      if s["tier"] in ("premium", "trial")
                      and airport_matches_subscriber(origin, s)]

      # Send instant email to matching premium/trial subscribers
      for sub in premium_subs:
        if self._email_send_count >= 90:  # Leave buffer below 100/day Gmail limit
          logger.warning("[ROUTER] Approaching Gmail 100/day limit, deferring sends")
          result["skipped_limit"] = True
          break

        # Build subject and HTML using existing email functions from deal_finder
        # Send via send_to_subscriber from mvp0_sender
        success = self._send_instant_email(sub, deal)
        if success:
          result["instant_emails"] += 1
          self._email_send_count += 1

      # SMS for mistake fares to premium subscribers with phone numbers
      if is_mistake:
        for sub in premium_subs:
          if sub.get("phone"):
            if send_sms_alert(sub["phone"], deal):
              result["sms_sent"] += 1

    # --- Queue for weekly digest ---
    # Great deals -> queued as free content
    # WOW/mistake deals -> also queued as FOMO teaser content
    if is_free_content:
      self._queue_for_digest(deal, is_teaser=False)
      result["digest_queued"] = True

    if is_premium_content:
      # Queue as expired/teaser for FOMO in free tier digest (FRML-01)
      self._queue_for_digest(deal, is_teaser=True)
      result["digest_queued"] = True

    return result

  _send_instant_email(self, subscriber: dict, deal: dict) -> bool:
    - Import send_to_subscriber from mvp0_sender
    - Build subject using format_alert_subject from alert.templates
    - Build HTML body using the existing format_destination_card_html and the full email template from deal_finder (build_email_content for a single deal)
    - Build plain body as fallback
    - Add historical price context for premium subscribers (SUBS-04):
      If deal has "observation_count" and deal has "z_score", add a line:
      "This price is {abs(z_score):.1f} standard deviations below the 90-day average ({observation_count} observations)"
    - Call send_to_subscriber(subscriber["email"], subject, html_body, plain_body)
    - Add 0.5s delay between sends (match existing pattern in mvp0_sender)
    - Return True/False

  _queue_for_digest(self, deal: dict, is_teaser: bool = False) -> bool:
    - Build digest queue entry dict:
      route: f"{deal['origin']}-{deal['dest']}"
      origin: deal["origin"]
      dest: deal["dest"]
      dest_name: deal["dest_name"]
      price_cents: int(deal.get("price", 0) * 100) if "price_cents" not in deal else deal["price_cents"]
      tier: deal.get("tier", "great")
      deal_data_json: (will be json.dumps'd by TursoClient method)
    - If is_teaser, set expired=1 in the deal dict (marks as FOMO teaser content)
    - Call self.db.queue_deal_for_digest(entry)
    - Return True/False

  route_deals(self, deals: list[dict]) -> dict:
    """Route multiple deals. Returns aggregate summary."""
    totals = {"instant_emails": 0, "sms_sent": 0, "digest_queued": 0, "skipped_limit": False}
    for deal in deals:
      result = self.route_deal(deal)
      totals["instant_emails"] += result["instant_emails"]
      totals["sms_sent"] += result["sms_sent"]
      if result["digest_queued"]:
        totals["digest_queued"] += 1
      if result["skipped_limit"]:
        totals["skipped_limit"] = True
    return totals
